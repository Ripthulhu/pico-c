cmake_minimum_required(VERSION 3.16)
project(pico_tiny C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(PICO_LOW_MEMORY "128-pixel scanline renderer with no draw list" OFF)
option(PICO_BUILD_X11 "Build the Linux/WSLg interactive reference host" ON)
option(PICO_BUILD_AUDIO "Build the fixed-memory mixer and host audio support" ON)
option(PICO_SANITIZE "Enable address and undefined-behavior checks" OFF)
option(PICO_EXPERIMENTAL_SMALL_POOLS "Use tested 192/64 pools below the conservative whole-game bound" OFF)

add_library(pico-core STATIC src/pico.c src/pico_render.c src/pico_ui.c generated/game_data.c)
target_include_directories(pico-core PUBLIC include generated)
if(PICO_EXPERIMENTAL_SMALL_POOLS)
  target_compile_definitions(pico-core PUBLIC PICO_MAX_INSTANCES=192 PICO_MAX_QUEUE=64)
else()
  target_compile_definitions(pico-core PUBLIC PICO_MAX_INSTANCES=320 PICO_MAX_QUEUE=128)
endif()

if(PICO_LOW_MEMORY)
  target_compile_definitions(pico-core PUBLIC PICO_STREAM_RENDER=1 PICO_RENDER_MAX_WIDTH=128)
else()
  target_compile_definitions(pico-core PUBLIC PICO_RENDER_MAX_WIDTH=3840)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pico-core PRIVATE -Wall -Wextra -Wpedantic -Werror -Wfatal-errors -Os -ffunction-sections -fdata-sections)
  if(PICO_SANITIZE)
    target_compile_options(pico-core PUBLIC -g -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pico-core INTERFACE -fsanitize=address,undefined)
  endif()
endif()

add_library(pico-host-common STATIC platform/host_common.c)
target_link_libraries(pico-host-common PUBLIC pico-core)
target_include_directories(pico-host-common PUBLIC platform)

add_executable(pico-headless platform/headless.c)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")

if(PICO_BUILD_AUDIO)
  set(PICO_AUDIO_MAX_VOICES 8 CACHE STRING "Fixed mixer voice count, from 1 to 32")
  if(NOT PICO_AUDIO_MAX_VOICES MATCHES "^([1-9]|[12][0-9]|3[0-2])$")
    message(FATAL_ERROR "PICO_AUDIO_MAX_VOICES must be an integer from 1 to 32")
  endif()
  add_library(pico-audio STATIC src/pico_audio.c)
  target_compile_definitions(pico-audio PUBLIC PICO_AUDIO_MAX_VOICES=${PICO_AUDIO_MAX_VOICES})
  target_link_libraries(pico-audio PUBLIC pico-core)
  add_library(pico-host-audio STATIC platform/host_audio.c)
  target_link_libraries(pico-host-audio PUBLIC pico-host-common pico-audio)
  target_compile_definitions(pico-host-audio PUBLIC PICO_HAS_AUDIO=1)
  target_link_libraries(pico-headless PRIVATE pico-host-audio)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(pico-audio PRIVATE -Wall -Wextra -Wpedantic -Werror -Os -ffunction-sections -fdata-sections)
    target_compile_options(pico-host-audio PRIVATE -Wall -Wextra -Wpedantic -Werror -Os)
  endif()
endif()

if(PICO_BUILD_X11 AND UNIX AND NOT APPLE)
  find_package(X11 QUIET)
  if(X11_FOUND)
    add_executable(pico-x11 platform/x11.c)
    target_link_libraries(pico-x11 PRIVATE pico-host-common X11::X11)
    if(PICO_BUILD_AUDIO)
      find_package(PkgConfig QUIET)
      if(PkgConfig_FOUND)
        pkg_check_modules(PULSE QUIET IMPORTED_TARGET libpulse)
      endif()
      if(PULSE_FOUND)
        find_package(Threads REQUIRED)
        target_sources(pico-x11 PRIVATE platform/audio_pulse.c)
        target_compile_definitions(pico-x11 PRIVATE PICO_HAVE_PULSE=1)
        target_link_libraries(pico-x11 PRIVATE pico-host-audio PkgConfig::PULSE Threads::Threads)
      else()
        message(STATUS "PulseAudio development package absent; X11 host will be silent")
      endif()
    endif()
  else()
    message(STATUS "X11 development package absent; building headless host only")
  endif()
endif()

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pico-host-common PRIVATE -Wall -Wextra -Wpedantic -Os)
  target_compile_options(pico-headless PRIVATE -Wall -Wextra -Wpedantic -Os)
  target_link_options(pico-headless PRIVATE -Wl,--gc-sections)
  if(TARGET pico-x11)
    target_compile_options(pico-x11 PRIVATE -Wall -Wextra -Wpedantic -Os)
    target_link_options(pico-x11 PRIVATE -Wl,--gc-sections)
  endif()
endif()
